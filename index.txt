<script>
function goTo(id){
  var pages=document.getElementsByClassName("page");

  for(var i=0;i<pages.length;i++){
    pages[i].classList.remove("active");
    pages[i].style.display="none";
  }

  var page=document.getElementById(id);

  if(page){
    page.style.display="block";
    page.classList.add("active");
    window.scrollTo(0,0);
  }
}

function openFinalSurprise(){
  goTo("final");
}

var hits=0;
var canHit=true;

var messages=[
"One heart caught! May happiness find you. ❤️",
"Two hearts! May all your dreams come true. 💗",
"Three hearts! Never lose your beautiful smile. 💖",
"Four hearts! May success always follow you. ✨",
"Five hearts! Birthday surprise unlocked, Akhi! 🎉"
];

function hitHeart(){
  if(hits>=5||!canHit)return;

  canHit=false;

  var arrow=document.getElementById("arrow");
  var heart=document.getElementById("targetHeart");

  arrow.classList.remove("shoot");
  void arrow.offsetWidth;
  arrow.classList.add("shoot");

  heart.style.visibility="hidden";

  setTimeout(function(){
    hits++;

    document.getElementById("gameScore").innerHTML="Hearts won: "+hits+" / 5";
    document.getElementById("hitMessage").innerHTML=messages[hits-1];

    if(hits===5){
      heart.style.display="none";
      arrow.style.display="none";
    }else{
      heart.style.visibility="visible";
      canHit=true;
    }
  },450);
}

var heartOpened=false;

var heartMessages=[
"❤️ May you always have a reason to smile, Akhi.",
"💗 May every dream in your heart come true, Akshuu.",
"💖 May your future shine brighter than the stars, Akhiluuu."
];

function showMessage(index){
  if(heartOpened)return;

  heartOpened=true;

  document.getElementById("heartMessage").innerHTML=heartMessages[index];
  document.getElementById("heartStatus").innerHTML="Your special heart is opened! The others are locked. 🔒";

  var hearts=document.getElementsByClassName("heartChoice");

  for(var i=0;i<hearts.length;i++){
    hearts[i].disabled=true;
    hearts[i].innerHTML=(i===index)?"💌":"🔒";
  }
}

var giftOpened=false;

var giftMessages=[
"🎉 Gift 1: A whole year filled with happiness and celebrations, Akhi!",
"❤️ Gift 2: Endless beautiful memories and smiles, Akshuu!",
"🌟 Gift 3: Success in every dream you follow, Akhiluuu!"
];

function openGift(index){
  if(giftOpened)return;

  giftOpened=true;

  document.getElementById("giftMessage").innerHTML=giftMessages[index];
  document.getElementById("giftStatus").innerHTML="Your special gift is opened! The other gifts are locked. 🔒";

  var gifts=document.getElementsByClassName("giftChoice");

  for(var i=0;i<gifts.length;i++){
    gifts[i].disabled=true;
    gifts[i].innerHTML=(i===index)?"💝":"🔒";
  }
}

function resetAll(){
  hits=0;
  canHit=true;
  heartOpened=false;
  giftOpened=false;

  document.getElementById("gameScore").innerHTML="Hearts won: 0 / 5";
  document.getElementById("hitMessage").innerHTML="Your first heart is waiting! 💕";

  document.getElementById("targetHeart").style.display="block";
  document.getElementById("targetHeart").style.visibility="visible";

  document.getElementById("arrow").style.display="block";
  document.getElementById("arrow").classList.remove("shoot");

  document.getElementById("nextGameButton").style.display="inline-block";

  document.getElementById("heartMessage").innerHTML="Choose your one heart! ✨";
  document.getElementById("heartStatus").innerHTML="";

  var hearts=document.getElementsByClassName("heartChoice");

  for(var i=0;i<hearts.length;i++){
    hearts[i].disabled=false;
    hearts[i].innerHTML=["❤️","💗","💖"][i];
  }

  document.getElementById("giftMessage").innerHTML="Your gifts are waiting! ❤️";
  document.getElementById("giftStatus").innerHTML="";

  var gifts=document.getElementsByClassName("giftChoice");

  for(var j=0;j<gifts.length;j++){
    gifts[j].disabled=false;
    gifts[j].innerHTML="🎁";
  }
}
